Given a pointed type $A$ and its pointed loop type $\loops A$,
we will define several auxiliary functions to be used in constructions
later.\footnote{Together these functions implement what is called a
H-space structure on $\loops A$, see \footcite{}.
Recall from \cref{sec:inftygps} that $A$ classifies an infinity group.
An H-structure on $\loops A$ can be viewed as an implementation
of some of the higher path algebra laws of the infinity group $A$.}

\begin{definition}\label{def:H-loop-type}
Let $A$ be a pointed type and $\loops A$ its pointed loop type.
Here and below, we use $\pt$ for $\pt_A$ and $\rfl$ for $\refl{\pt_A}$.
Define:
\begin{align*}
\mu_l &: \prod_{p:\loops A} (p\eqto(\rfl \cdot p),
~~ \mu_l(p)\defeq\refl{p} \\
\mu_r &: \prod_{p:\loops A} (p\eqto(p \cdot \rfl)), 
~~ \mu_r(\rfl)\defeq\refl{\rfl}, \text{ using path induction}\\
\mu_{lr} &: \mu_l (\rfl) \eqto \mu_r (\rfl), 
~~ \mu_{lr}\defeq  \refl{\refl{\rfl}}, \text{ called \emph{coherence}}.
\qedhere
\end{align*}
\end{definition}

\begin{construction}\label{con:ev-section-loopsA}
Let $A$ be a pointed type and $\loops A$ its pointed loop type.
Here and below, we use $\pt$ for $\pt_A$ and $\rfl$ for $\refl{\pt_A}$.
Let $\ev: (\loops A \to\loops A,\id_{\loops A})\ptdto (\loops A,\rfl)$ 
be the evaluation map that sends $f$ to $f(\rfl)$ and is pointed 
by $\ev_\pt\defeq\refl{\rfl}$. 
Then $\ev$ has a pointed section, that is, a pointed 
map $s: (\loops A,\rfl)\ptdto(\loops A \to\loops A,\id_{\loops A})$
such that we have an identification of $(\id_{\loops A},\refl{\rfl})$ with
${\ev}\circ{s}$.
\end{construction}

\begin{implementation}{con:ev-section-loopsA}
Let $\mu_l,\mu_r,\mu_{lr}$ be given as in \cref{def:H-loop-type}.
Define $s$ by $s(p)\defeq (p\cdot\blank)$ and point $s$ by
$s_\pt\defeq\inv\ptw(\mu_l): \id_{\loops A}\eqto s(\rfl)$.

We use the inverse equivalence of $\ptw_*$ from \cref{con:identity-ptd-maps}
to get the desired identification of $(\id_{\loops A},\refl{\rfl})$ with
${\ev}\circ{s}$. This reduces our task to, first, constructing a function
$h: \prod_{p:\loops A} (p \eqto \ev(s(p))$.
As $\ev(s(p))\jdeq p\cdot\rfl$, we can take $\mu_r$ for $h$.
Second, we must fill the diagram in \cref{fig:ev-section-loopsA}.
The pointing path of $\id_{\loops A}$ is $\refl{\rfl}$.
By \cref{def:pointedtypes} the pointing path $({\ev}\circ{s})_\pt$ is 
$(\ap{\ev}(s_\pt)\cdot \ev_\pt) \jdeq 
(\ap{\ev}(\inv\ptw(\mu_l))\cdot \refl{\rfl})$, 
which can be identified\footnote{%
By induction on $q: f \eqto_{X\eqto X} g$ one easily constructs
an identification $\ap{\ev_x}(q) \eqto \ptw(q)(x)$ for any
type $X$, element $x:X$, functions $f,g:X\to X$, and
$\ev_x$ evaluation at $x$.} 
with $\ptw(\inv\ptw(\mu_l))(\rfl)$ and hence with $\mu_l(\rfl)$.
Now use coherence, $\mu_{lr}$.
\end{implementation}

Note that, in the above construction, 
$\loops(\ev) :(\id_{\loops A}\eqto\id_{\loops A})\to(\rfl\eqto\rfl)$
maps any $q:(\id_{\loops A}\eqto\id_{\loops A})$ to
$\inv{\refl{\rfl}}\cdot\ap{\ev}(q)\cdot\refl{\rfl}$,
which can be identified with $\ptw(q)(\rfl)$, 
for any $q:(\id_{\loops A}\eqto\id_{\loops A})$.

\begin{corollary}\label{cor:Omega-ev-section}
\cref{con:ev-section-loopsA} induces an identification of
$\loops({\ev}\circ{s})$ with $\id_{\rfl\eqto\rfl}$, so the unpointed
map $\loops(s): (\pt\eqto\pt)\to (\id_{\loops A}\eqto\id_{\loops A})$ 
is a section of $\loops(\ev)$, so of $q\mapsto\ptw(q)(\rfl)$.
\end{corollary}

\begin{definition}\label{def:abshom_!}
  Given groups $G$, $H$ and an abstract homomorphism
  $\phi:\absHom(\abstr(G),\abstr(H))$, we define the map $\phi_!$
  from $G$-sets to $H$-sets as follows. 
  For any $G$-set $X:\BG\to\Set$ and $w:\BH$, define
  \[
    \phi_!X(w)\defequi \bigl((\sh_H \eqto w) \times_\phi X(\sh_G)\bigr)
  \]
  to be the set quotient of $(\sh_H\eqto w)\times X(\sh_G)$ modulo
  the equivalence relation $(p,x)\sim(q,y)$ if there exists a $g:\USymG$
  such that $p=q\phi(g)$ and $g\cdot_X x = y$.
\end{definition}

\begin{lemma}\label{lem:abshom_!}
  With $\phi_!$ as in \cref{def:abshom_!}, the map 
  $\eta_\phi:\phi_!\princ G \equivto \princ H$ sending, for all $w:\BH$,
  $[(p,x)]: (\sh_H \eqto w) \times_\phi \USymG$ to $p\phi(x):(\sh_H\eqto w)$,
  is a well defined (fiberwise) equivalence. Consequently,
  $(\phi_!,\inv{\eta_\phi})$ is a pointed map from
  $(\typetorsor_G,\princ G)$  to $(\typetorsor_H,\princ H)$.
\end{lemma}
\begin{proof}
First we show that $\eta_\phi$ respects the equivalence relation.
Let $(p,x)\sim(q,y)$ with $p,q:(\sh_H \eqto w)$ and $x,y:\USymG$.
Then there exists a $g:\USymG$ such that $p=q\phi(g)$ and $g\cdot_X x = y$.
Now, $p\phi(x)=q\phi(gx) = q\phi(y)$, so $\eta_\phi$
is indeed well defined.
It is also clearly a surjection. So it remains to prove that $\eta_\phi$
is injective. Assume $(p,x)$ and $(q,y)$ are such that
$p\phi(x) = q\phi(y)$. Then $p= q\phi(y\inv{x})$ and 
$y\inv{x}\cdot_X x = y$. 
Hence $(p,x)\sim(q,y)$, so their classes are equal.
This shows that $\eta_\phi$ is injective, and completes the proof.
\end{proof}


\begin{xca}\label{xca:phi_!-on-paths}
Recall \cref{def:abshom_!} and show that $\phi_!\pi(w)$ maps $[(p,x)]$ 
in $\phi_!X(w)$ to $[(p,\pi_{\sh_G}(x))]$ in $\phi_!X'(w)$,
for any path $\pi: X \eqto X'$ and $w:\BH$.
Then prove that $\loops(\phi_!,\inv{\eta_\phi})$ sends 
$\preinv_{\blank}(g)$ to $\preinv_{\blank}(\phi(g))$.
Hint: recall \cref{def:loops-map} and start by 
making $\inv{\eta_\phi}$ explicit.
\end{xca}
